// Demonstrating Java OOP, Encapsulation, Collections & Stream API
package com.sadikali.portfolio;

import java.util.ArrayList;
import java.util.List;
import java.util.stream.Collectors;

public class ProjectManager {
    private final List<Project> projects = new ArrayList<>();

    // Record demonstrating modern Java data modeling
    public record Project(String title, String language, boolean isDeployed) {}

    public void addProject(String title, String language, boolean isDeployed) {
        projects.add(new Project(title, language, isDeployed));
    }

    public List<Project> getDeployedProjectsByTech(String tech) {
        return projects.stream()
            .filter(Project::isDeployed)
            .filter(p -> p.language().equalsIgnoreCase(tech))
            .collect(Collectors.toList());
    }

    public static void main(String[] args) {
        ProjectManager pm = new ProjectManager();
        pm.addProject("Motor Doctor", "JavaScript", true);
        pm.addProject("Student Portal", "Java", true);
        pm.addProject("Web Scraper", "Python", true);

        System.out.println("Java Projects count: " + pm.getDeployedProjectsByTech("Java").size());
    }
}

# Demonstrating Pythonic Automation, REST API Consumption & Error Handling
import urllib.request
import json
from typing import List, Dict, Any

class GitHubPortfolioTracker:
    """Fetches and summarizes public repositories using clean Python OOP."""

    BASE_URL: str = "https://api.github.com/users/{username}/repos"

    def __init__(self, username: str) -> None:
        self.username = username
        self.endpoint = self.BASE_URL.format(username=username)

    def fetch_repositories(self) -> List[Dict[str, Any]]:
        try:
            request = urllib.request.Request(
                self.endpoint,
                headers={"User-Agent": "Python-Portfolio-Tracker"}
            )
            with urllib.request.urlopen(request, timeout=10) as response:
                return json.loads(response.read().decode("utf-8"))
        except Exception as exc:
            print(f"Error fetching repositories: {exc}")
            return []

    def get_tech_breakdown(self) -> Dict[str, int]:
        repos = self.fetch_repositories()
        stats: Dict[str, int] = {}
        for repo in repos:
            lang = repo.get("language") or "Other"
            stats[lang] = stats.get(lang, 0) + 1
        return stats

if __name__ == "__main__":
    tracker = GitHubPortfolioTracker("sadikali67011-art")
    breakdown = tracker.get_tech_breakdown()
    print("Repository Tech Distribution:", breakdown)

// Demonstrating Modern ES6+, Async/Await, Web API & DOM State Management
class InteractivePortfolio {
  constructor() {
    this.theme = localStorage.getItem("portfolio-theme") || "dark";
    this.init();
  }

  init() {
    document.documentElement.setAttribute("data-theme", this.theme);
    this.attachScrollListener();
  }

  async copyToClipboard(text, successCallback) {
    try {
      await navigator.clipboard.writeText(text);
      if (typeof successCallback === "function") {
        successCallback();
      }
    } catch (err) {
      console.error("Clipboard write failed:", err);
    }
  }

  attachScrollListener() {
    const backToTop = document.querySelector("#back-to-top");
    window.addEventListener("scroll", () => {
      const isScrolled = window.scrollY > 400;
      backToTop?.classList.toggle("visible", isScrolled);
    }, { passive: true });
  }
}

// Instantiate application cleanly on DOM load
document.addEventListener("DOMContentLoaded", () => new InteractivePortfolio());
